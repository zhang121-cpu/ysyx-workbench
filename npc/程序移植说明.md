# NPC 移植 NEMU 框架 · 最终完整文档（v3.8）

## 〇、设计判据（先看这个，后面所有取舍都由此推出）

nemu 里"CPU 状态"唯一定义在全局变量 `CPU_state cpu`。所有文件按它和 `cpu` 的关系分三类：

```
【状态消费者】读 cpu            【状态来源】写 cpu
  sdb / expr / watchpoint         解释器 (isa_exec_once)
  itrace / ftrace / mtrace        设备 (device/*)
  difftest 比较                   内存后端 (paddr.c 是共享存储)
        │                              │
        └────── 中间：取寄存器访问器 ──────┘
              isa_reg_display / isa_reg_str2val
```

> 🎯 **判据 1**：读 CPU 状态的 → 拷；写 CPU 状态的（指令译码 inst.c、设备 device/*）→ 不拷（NPC 里这一层是 RTL）。
> 🎯 **判据 2**：描述"有哪些选项"的（Kconfig）→ 保留 nemu 骨架，只删 npc 物理上不存在的；实现"主程序怎么构建/运行"的 → 看引擎是否匹配；独立工具及其规则 → 拷。
> 🎯 **判据 3**：机器生成物（`include/config/`、`include/generated/`）→ 永远不拷，.gitignore 排除。

**核心机制——影子 CPU**：npc 侧保留全局 `CPU_state cpu`（在重写的 cpu-exec.c 里定义），由两路 DPI 事件实时同步：GPR.v 写回 → `npc_sync_gpr(idx,val)`；top.v 每拍 → `npc_commit_inst(pc,inst,dnpc)`。于是 nemu 所有"读 cpu"的代码（sdb/expr/reg.c/difftest/paddr.c 的报错）**几乎一个字符都不用改**。

> ⚠️ **贯穿全文的移植提醒约定**：Kconfig 里凡标了 `【移植提醒】` 的注释，都是"这个符号在 nemu 里存在、但在 npc 里被保留/改名/改义"的地方——读代码时看到它们就知道"这是移植留下的痕迹"。

#### 6.1 `src/cpu/cpu-exec.c`（重写：影子 CPU + exec_once 换成时钟驱动 + DPI 事件；ebreak 走完整流程；新增 MAX_SIM_CYCLES）

```c
#include <cpu/cpu.h>
#include <cpu/decode.h>
#include <cpu/difftest.h>
#include <locale.h>
#include <npc_api.h>

#define MAX_INST_TO_PRINT 10

#ifdef CONFIG_ITRACE
#define IRINGBUF_SIZE 16
typedef struct {
  uint32_t pc;
  uint32_t inst;
  char disasm[128];
} ItraceStruct;

ItraceStruct iringbuf[IRINGBUF_SIZE];
int iringbuf_index = 0;
#endif

#ifdef CONFIG_ITRACE
void iringbuf_display() {
  printf("\n---- iringbuf (last %d instructions) ----\n", IRINGBUF_SIZE);
  for (int i = 0; i < IRINGBUF_SIZE; i++) {
    if (i == ((iringbuf_index == 0) ? (IRINGBUF_SIZE-1) : (iringbuf_index-1))) printf("--> ");
    else printf("    ");
    printf("0x%08x: %08x\t%s\n", iringbuf[i].pc, iringbuf[i].inst, iringbuf[i].disasm);
  }
  printf("----------------------------------------\n");
}
#endif

/* DUT 寄存器影子：由 DPI 事件维护。nemu 拷来的代码全部读它，故零修改。 */
CPU_state cpu = {};
uint64_t g_nr_guest_inst = 0;
static uint64_t g_timer = 0; // unit: us
static bool g_print_step = false;

int wp_check();

/* ============ DPI 事件处理（由 vsrc 在时钟上升沿调用） ============ */

static Decode s_commit;
static bool commit_valid = false;

/* GPR 写回事件（vsrc/GPR.v）。x0 恒 0 在此保证。 */
void npc_sync_gpr(uint32_t idx, uint32_t val) {
  if (idx != 0) cpu.gpr[idx] = val;
}

#ifdef CONFIG_FTRACE
void ftrace_call(uint32_t pc, uint32_t target);
void ftrace_ret(uint32_t pc);
static void ftrace_detect(uint32_t pc, uint32_t inst, uint32_t dnpc) {
  uint32_t opcode = inst & 0x7f;
  uint32_t rd     = (inst >> 7) & 0x1f;
  uint32_t rs1    = (inst >> 15) & 0x1f;
  if (opcode == 0x6f) {                      /* jal */
    if (rd != 0) ftrace_call(pc, dnpc);      /* rd!=x0 才是调用（对齐 nemu） */
  } else if (opcode == 0x67) {               /* jalr */
    if (rd != 0) ftrace_call(pc, dnpc);      /* rd!=x0 才是调用 */
    else if (rd == 0 && rs1 == 1) ftrace_ret(pc);  /* rd==x0 且 rs1==ra 才是返回（对齐 nemu，去掉 imm_i==0） */
  }
}
#endif

/* 指令提交事件（vsrc/top.v）：RTL 中一条指令执行完毕（含 ebreak） */
void npc_commit_inst(uint32_t pc, uint32_t inst, uint32_t dnpc) {
  s_commit.pc   = pc;
  s_commit.snpc = pc + MUXDEF(CONFIG_ISA_x86, 8, 4);
  s_commit.dnpc = dnpc;
  s_commit.isa.inst = inst;
  cpu.pc = dnpc;
  commit_valid = true;              /* 所有指令都设 true（含 ebreak，对齐 nemu） */
  IFDEF(CONFIG_FTRACE, ftrace_detect(pc, inst, dnpc));
}

/* ============ 以下与 nemu 的 cpu-exec.c 相同，仅 exec_once 内部不同 ============ */

static void trace_and_difftest(Decode *_this, vaddr_t dnpc) {
#ifdef CONFIG_ITRACE_COND
  if (ITRACE_COND) { log_write("%s\n", _this->logbuf); }
#endif
  if (g_print_step) { IFDEF(CONFIG_ITRACE, puts(_this->logbuf)); }
  IFDEF(CONFIG_DIFFTEST, difftest_step(_this->pc, dnpc));

#ifdef CONFIG_WATCHPOINT
  int symbol = wp_check();
  if (symbol) {
    if (nemu_state.state != NEMU_END)
      nemu_state.state = NEMU_STOP;
  }
#endif
}

static void exec_once(Decode *s, vaddr_t pc) {
  commit_valid = false;
  while (!commit_valid) {  //流水线下单指令多周期起作用
    npc_single_cycle();
  #ifdef CONFIG_WAVE_RECORD
    npc_wave_record();
  #endif
  }
  *s = s_commit;

#ifdef CONFIG_ITRACE
  char *p = s->logbuf;
  p += snprintf(p, sizeof(s->logbuf), FMT_WORD ":", s->pc);
  int ilen = s->snpc - s->pc;    /* 从 snpc-pc 算指令长度（恢复 nemu 原版写法，扩展性） */
  int i;
  uint8_t *inst = (uint8_t *)&s->isa.inst;
#ifdef CONFIG_ISA_x86
  for (i = 0; i < ilen; i ++) {      /* x86：正序 */
#else
  for (i = ilen - 1; i >= 0; i --) { /* riscv：倒序 */
#endif
    p += snprintf(p, 4, " %02x", inst[i]);
  }
  int ilen_max = MUXDEF(CONFIG_ISA_x86, 8, 4);    /* x86 最长 8 字节，riscv 4 */
  int space_len = ilen_max - ilen;
  if (space_len < 0) space_len = 0;
  space_len = space_len * 3 + 1;
  memset(p, ' ', space_len);
  p += space_len;

  void disassemble(char *str, int size, uint64_t pc, uint8_t *code, int nbyte);
  disassemble(p, s->logbuf + sizeof(s->logbuf) - p,
      MUXDEF(CONFIG_ISA_x86, s->snpc, s->pc), (uint8_t *)&s->isa.inst, ilen);

  //将指令存入iringbuf中
  iringbuf[iringbuf_index].pc = s->pc;
  iringbuf[iringbuf_index].inst = s->isa.inst;
  disassemble(iringbuf[iringbuf_index].disasm, sizeof(iringbuf[iringbuf_index].disasm),
      MUXDEF(CONFIG_ISA_x86, s->snpc, s->pc), (uint8_t *)&s->isa.inst, ilen);
  iringbuf_index = (iringbuf_index + 1) % IRINGBUF_SIZE;
#endif
}

static void execute(uint64_t n) {
  Decode s;
  for (; n > 0; n--) {
    exec_once(&s, cpu.pc);
    g_nr_guest_inst++;               //一个用于记录客户指令的计数器
    trace_and_difftest(&s, cpu.pc);
  #ifdef CONFIG_MAX_SIM_CYCLES_ENABLE    //周期数达到上限时自动结束模拟
    if (g_nr_guest_inst > CONFIG_MAX_SIM_CYCLES){
      Log("total cycles has reached the limit");
      nemu_state.state = NEMU_QUIT;
    }
  #endif
    if (nemu_state.state != NEMU_RUNNING) break;
  }
}

static void statistic() {
  setlocale(LC_NUMERIC, "");
#define NUMBERIC_FMT "%'" PRIu64
  Log("host time spent = " NUMBERIC_FMT " us", g_timer);
  Log("total guest instructions = " NUMBERIC_FMT, g_nr_guest_inst);
  if (g_timer > 0) Log("simulation frequency = " NUMBERIC_FMT " inst/s", g_nr_guest_inst * 1000000 / g_timer);
  else Log("Finish running in less than 1 us and can not calculate the simulation frequency");
}

void assert_fail_msg() {
  isa_reg_display();
#ifdef CONFIG_ITRACE
  iringbuf_display();
#endif
  statistic();
}

void cpu_exec(uint64_t n) {
  g_print_step = (n < MAX_INST_TO_PRINT);
  switch (nemu_state.state) {
    case NEMU_END: case NEMU_ABORT: case NEMU_QUIT:
      printf("Program execution has ended. To restart the program, exit and run again.\n");
      return;
    default: nemu_state.state = NEMU_RUNNING;
  }

  uint64_t timer_start = get_time();

  execute(n);

  uint64_t timer_end = get_time();
  g_timer += timer_end - timer_start;

  switch (nemu_state.state) {
    case NEMU_RUNNING: nemu_state.state = NEMU_STOP; break;

    case NEMU_ABORT:
#ifdef CONFIG_ITRACE
      iringbuf_display();
#endif
      Log("npc: %s at pc = " FMT_WORD, ANSI_FMT("ABORT", ANSI_FG_RED), nemu_state.halt_pc);
      statistic();
      break;

    case NEMU_END:
      if (nemu_state.halt_ret != 0) {
#ifdef CONFIG_ITRACE
        iringbuf_display();
#endif
      }
      Log("npc: %s at pc = " FMT_WORD,
          (nemu_state.halt_ret == 0 ? ANSI_FMT("HIT GOOD TRAP", ANSI_FG_GREEN) :
            ANSI_FMT("HIT BAD TRAP", ANSI_FG_RED)),
            nemu_state.halt_pc);
      statistic();
      break;

    case NEMU_QUIT: statistic();
  }
}
```

```c
  #ifdef CONFIG_HAS_TIMER
```

```cpp
//接着如果检测到上升沿，计算当前的 nba 全组合链

extern VerilatedContext *contextp;
extern Vtop *top;
extern VerilatedFstC *tfp;

static bool in_clk0_eval = false;

bool npc_in_clk0_eval(void) {
    return in_clk0_eval;
}

void npc_single_cycle(void) {
    in_clk0_eval = true; top->clk = 0; top->eval();   // 第 1 次 eval，记录
    in_clk0_eval = false; top->clk = 1; top->eval();  // 第 2 次 eval，不记录
}

#ifdef CONFIG_WAVE_RECORD

    tfp->dump(contextp->time());
    contextp->timeInc(1);
}
#endif

void npc_reset(int n) {
    top->rst = 1;
    for (int i = 0; i < n; i++) npc_single_cycle();
    top->rst = 0;
}
```

3. **默认目标是 `app`**：`make build` 不存在（`No rule to make target 'build'`）；直接 `make` 或 `make app`。
4. **`npc_sync_gpr` 的 x0 保护在 C 侧**（`idx != 0`）。
5. **设备访问必须 `difftest_skip_ref()`**（幂等，安全）。
6. **组合逻辑 DPI 每次 eval 都调**（一拍两次）：mtrace/dtrace 的 READ 重复问题已用相位门控修复。
7. **disasm.c 运行时 dlopen 相对路径**：必须从 npc 运行 `./build/Vtop`，capstone 已拷（含 `libcapstone.so.5`）。
8. **`find -L ./src` 要求从 npc 运行 make**。
9. **`src/isa/filelist.mk` 已含 `$(GUEST_ISA)` 逻辑**：不要在 `src/filelist.mk` 重复加 isa 目录。
10. **【移植提醒】`CONFIG_DEVICE` 永不定义**：npc 的设备总开关是 **`CONFIG_NPC_DEVICE`**（Kconfig 顶部有显眼注释）。子开关 `HAS_SERIAL`/`HAS_TIMER` 只在 npc_dpi.c 使用。
11. **【移植提醒】`NPC_DEVICE` 默认关**：设备默认不工作——程序若 `putchar`/读 RTC 会 `out_of_bound` panic。要么程序先不碰设备，要么 `make menuconfig` 打开 "Devices (NPC)"。
12. **【移植提醒】`TARGET_SHARE`/`TARGET_AM` 已删**：npc 不做 REF .so / AM 应用。代码里 `TARGET_AM` 全是"开了走 AM 路径，没开走正常路径"，不定义就走正常路径，安全。`depends on !TARGET_AM` 是有意保留的 nemu 原文残留（条件恒真）。
13. **dtrace 的位置**：`DTRACE` 放 `src/device/Kconfig` 末尾（`NPC_DEVICE` 块内，对齐 nemu）；触发点在 npc_dpi.c；实现来自拷来的dtrace.c （日志写到 `build/npc-log-dtrace.txt`）；monitor.c 里 `IFDEF(CONFIG_DTRACE, init_dtrace)` 拷来即带。
14. **ebreak 走完整流程**：top.v 里 ebreak 也调 `npc_commit_inst`（计入、记录）；ebreak跳过difftest 比较这一任务由 `set_nemu_state` 内部的 `difftest_skip_ref()` 完成（hostcall.c:22），`npc_trap` 只调 set_nemu_state ，这和 nemu 逻辑一致。
15. **设备地址约定**：当前情况下， npc 为`SERIAL=0x10000000 / RTC=0x10000004`（由 `src/device/Kconfig` 的 `SERIAL_MMIO`/`RTC_MMIO` 提供）； nemu 为 `0xa00003f8 / 0xa0000048`（由 `src/device/Kconfig` 的 `SERIAL_MMIO`/`RTC_MMIO` 提供）。RTC "先读高、再读低"的程序逻辑要求与 nemu 一致。
16. **DiffTest 的 RVE 两边一致**否则 `DIFFTEST_REG_SIZE` 错位（这一条还没看懂，RVE应该是在 src/isa/riscv32/Kconfig 中，但目前都默认为关，也没出问题）。
17. **`tools/kconfig` 需 `flex`/`bison`/`ncurses`**（npc 能 menuconfig 即已具备）。
18. **build.mk 拆分**：纯 C 编译（通用）+ Verilator 专用，避免 tools 子项目触发 Verilator。
19. **`compile_git` 位置**：`$(BINARY):: compile_git`（与 nemu 对齐）。
20. **clean 分层**：`clean` 在 build.mk（`rm -rf build/`，主程序/tools 通用）；`clean-tools`、`clean-all = clean + distclean + clean-tools` 在 native.mk。
21. **变量统一**：`BINARY` 统一命名，替代 `SIM_BIN`，与 nemu 对齐。
22. **智能区分**：`WORK_DIR` 判断是否等于 `NPC_HOME`，自动区分主程序/子程序。
23. **Verilator 分离**：去掉 `--build`，分离生成和编译，支持增量编译。
24. **波形文件管理**：每次编译前删除 wave.fst，避免误用。
25. **mtrace 区分**：`vaddr_ifetch`（取指，不记录）vs `vaddr_read`（数据访存，记录）。
26. **MAX_SIM_CYCLES**：`MAX_SIM_CYCLES_ENABLE` + `MAX_SIM_CYCLES`（最大周期数限制）。
27. **WAVE_RECORD**：波形记录开关。
28. **波形文件路径**：从 `log_file` 推导 wave.fst 路径（`init_wave` 在 monitor.c 里随 log 初始化后调用；driver.cpp 的 main 里复位后先记录一拍）。

## 八、移植完成后与 nemu 完全一致的交互

```
(npc) si 5        ← 每条指令 = 驱动时钟直到一次提交，itrace 同步打印
(npc) info r      ← 读影子 cpu（由 GPR.v 写回事件实时同步）
(npc) x 4 0x80000000   ← vaddr_read → paddr_read → 与 RTL 同一块 pmem
(npc) p $a0 + 4   ← expr.c 零修改
(npc) w $pc == 0x80000010   ← 每指令 wp_check
(npc) c           ← 跑到底，ebreak → HIT GOOD TRAP + 统计信息
```

## 一句话总结

> **源代码侧**用"影子 CPU"让 nemu 的 sdb/trace/difftest 近乎零修改拷贝；**构建侧**用 nemu 的四层 `.mk` 分工（build.mk 纯 C 编译 + verilator.mk Verilator 专用 + native.mk run/gdb + config.mk menuconfig），默认目标 `app`、`$(BINARY)` 统一命名、`NPC_HOME` 由环境提供（与 nemu 的 `NEMU_HOME` 同款约定）；**ISA 层**保留完整骨架（5 个 ISA 全拷，choice 原样）；**配置层**用 `CONFIG_NPC_DEVICE` 取代 `CONFIG_DEVICE`（后者永不定义），`TARGET_SHARE`/`TARGET_AM` 删除（仅  `depends on !TARGET_AM` 作为 nemu 原文残留保留），DiffTest REF 默认 NEMU（KVM 预留），新增 `MAX_SIM_CYCLES`/`WAVE_RECORD`。所有"移植留下的痕迹"用 `【移植提醒】` 注释标出。
